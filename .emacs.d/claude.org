* Agent shell
Agent shell is an emacs package that provides a native client to the underlying Claude Code agent.
** Enable global revert mode
This ensures that files =claude= modifies automatically update in Emacs:
#+begin_src elisp :results none
(global-auto-revert-mode)
#+end_src
** Install claude code ACP support
Support for the Agent Client Protocol needs to be installed first:
#+begin_src bash :results output verbatim :results none
npm install -g @agentclientprotocol/claude-agent-acp
#+end_src
** Install agent shell
#+begin_src elisp :results none
(use-package agent-shell :ensure t)
(require 'agent-shell)
#+end_src
** Inherit agent shell environment from emacs
#+begin_src elisp :results none
(setq agent-shell-anthropic-claude-environment
      (agent-shell-make-environment-variables :inherit-env t))
#+end_src
** Configure agent shell for claude code
#+begin_src elisp :results none
(setq agent-shell-anthropic-authentication
      (agent-shell-anthropic-make-authentication :login t))
#+end_src
** Hide line numbers in agent-shell-mode
#+begin_src elisp :results none
(defun ak/agent-shell-mode-hook ()
  (display-line-numbers-mode -1))

(add-hook 'agent-shell-mode-hook #'ak/agent-shell-mode-hook)
#+end_src
** Global key bindings
#+begin_src elisp :results none
(define-key projectile-mode-map (kbd "C-c p c c") #'agent-shell-anthropic-start-claude-code)
(define-key projectile-mode-map (kbd "C-c p c s") #'agent-shell-send-region)
#+end_src
** Agent shell customisations
#+begin_src elisp :results none
(setq agent-shell-header-style "text")
#+end_src
